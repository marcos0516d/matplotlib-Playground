plt.subplots() - Subplots con grilla

Crea multiples graficos en una sola figura usando filas y columnas.

# Subplots 2x2
fig, axes = plt.subplots(2, 2, figsize=(10, 8))

axes[0, 0].plot([1, 2, 3], [1, 4, 9])
axes[0, 0].set_title('Lineas')

axes[0, 1].bar(['A', 'B', 'C'], [3, 7, 2])
axes[0, 1].set_title('Barras')

axes[1, 0].scatter([1, 2, 3], [3, 1, 2])
axes[1, 0].set_title('Dispersion')

axes[1, 1].hist([1, 2, 2, 3, 3, 3])
axes[1, 1].set_title('Histograma')

plt.tight_layout()

Subplots verticales

Organiza graficos en columnas usando ncols.

# 3 graficos en fila
fig, axes = plt.subplots(1, 3, figsize=(15, 4))

axes[0].plot([1, 2, 3], [1, 4, 9])
axes[1].bar(['A', 'B'], [5, 3])
axes[2].scatter([1, 2, 3], [2, 1, 3])

plt.tight_layout()

Subplots cuadriculados

Crea grillas de graficos con nrows y ncols.

# Grilla 3x2
fig, axes = plt.subplots(nrows=3, ncols=2, figsize=(12, 10))

# Acceder a cada grafico por indice
for i in range(3):
    for j in range(2):
        axes[i, j].plot([1, 2, 3], [i+j, i*j, i-j])

plt.tight_layout()

plt.style.use() - Estilos incorporados

Usa estilos predefinidos para cambiar la apariencia global de tus graficos.

# Ver estilos disponibles
print(plt.style.available)

# Usar un estilo
plt.style.use('ggplot')
plt.plot([1, 2, 3], [1, 4, 9])
plt.show()

# Estilos populares: 'ggplot', 'seaborn', 'dark_background', 'fivethirtyeight'

rcParams personalizado

Configura fuentes, colores y tamanios globalmente para todos tus graficos.

# Configurar parametros globales
plt.rcParams['font.size'] = 12
plt.rcParams['figure.figsize'] = (10, 6)
plt.rcParams['axes.titlesize'] = 16
plt.rcParams['axes.labelsize'] = 14
plt.rcParams['lines.linewidth'] = 2

# Restablecer valores por defecto
plt.rcParams.update(plt.rcParamsDefault)

Colormaps

Aplica mapas de colores a tus graficos para representar valores o categorias.

# Colormap en scatter
import numpy as np
x = np.random.rand(50)
y = np.random.rand(50)
colores = np.random.rand(50)

plt.scatter(x, y, c=colores, cmap='viridis')
plt.colorbar()

# Colormaps: 'viridis', 'plasma', 'coolwarm', 'RdYlGn', 'jet'

Histograma avanzado

Personaliza bins, densidad y acumulado en histogramas.

datos = np.random.normal(0, 1, 1000)

# Histograma con densidad
plt.hist(datos, bins=30, density=True, alpha=0.7, color='steelblue')

# Histograma acumulado
plt.hist(datos, bins=30, cumulative=True, alpha=0.7, color='coral')

plt.boxplot() - Boxplot

Visualiza distribuciones y outliers con diagramas de caja.

# Boxplot simple
datos = [np.random.normal(0, 1, 100) for _ in range(4)]
plt.boxplot(datos, labels=['A', 'B', 'C', 'D'])

# Con personalizacion
plt.boxplot(datos, patch_artist=True,
            boxprops=dict(facecolor='lightblue'))

plt.violinplot() - Violin plot

Combina boxplot y KDE para mostrar la forma completa de la distribucion.

# Violin plot
datos = [np.random.normal(0, 1, 100) for _ in range(4)]
plt.violinplot(datos, showmeans=True, showmedians=True)

# Con posiciones personalizadas
plt.violinplot(datos, positions=[1, 2, 3, 4])

plt.imshow() - Heatmap

Crea mapas de calor para visualizar matrices de datos.

# Heatmap simple
import numpy as np
data = np.random.rand(5, 5)
plt.imshow(data, cmap='hot', interpolation='nearest')
plt.colorbar()

# Heatmap con anotaciones
plt.imshow(data, cmap='coolwarm')
for i in range(5):
    for j in range(5):
        plt.text(j, i, f'{data[i,j]:.2f}', ha='center', va='center')

plt.errorbar() - Graficos de error

Muestra incertidumbre en tus datos con barras de error.

# Error bars
x = [1, 2, 3, 4, 5]
y = [2, 4, 5, 4, 5]
errores = [0.5, 0.3, 0.8, 0.2, 0.4]

plt.errorbar(x, y, yerr=errores, fmt='o', capsize=5, capthick=2)

# Con error horizontal
plt.errorbar(x, y, xerr=errores, fmt='s', ecolor='red')

Barras con error bars

Combina barras con barras de error para comparaciones con incertidumbre.

# Barras con error
categorias = ['A', 'B', 'C', 'D']
valores = [23, 45, 12, 67]
errores = [3, 5, 2, 7]

plt.bar(categorias, valores, yerr=errores, capsize=8,
        color='steelblue', alpha=0.7)

plt.stackplot() - Area apilada

Crea graficos de area apilada para mostrar la composicion de un todo.

# Area apilada
x = [1, 2, 3, 4, 5]
y1 = [1, 2, 3, 4, 5]
y2 = [2, 3, 2, 3, 4]
y3 = [3, 2, 4, 3, 2]

plt.stackplot(x, y1, y2, y3, labels=['Serie A', 'Serie B', 'Serie C'],
              colors=['#3498db', '#2ecc71', '#e74c3c'], alpha=0.7)
plt.legend(loc='upper left')
